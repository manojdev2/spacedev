Database Design Principles

  • Multi-Tenancy: All user data is scoped to organizations
  • Row Level Security: Every table has RLS policies
  • Soft Relationships: User IDs reference auth.users without foreign keys
  • JSON for Flexibility: Complex data (opening_hours, preferences) stored as JSONB

Key Design Patterns

1. Custom Hook Pattern

All data fetching is abstracted into custom hooks:

export function usePublicLocations() {
  return useQuery({
    queryKey: ['locations', 'public'],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('locations')
        .select('*')
        .eq('is_active', true);
      if (error) throw error;
      return data;
    }
  });
}

2. Component Composition

Page
└── Layout
    ├── Header
    ├── Main Content
    │   ├── Feature Component
    │   │   ├── UI Component (shadcn)
    │   │   └── UI Component (shadcn)
    │   └── Feature Component
    └── Footer

3. Optimistic Updates

useMutation({
  onMutate: async (newData) => {
    await queryClient.cancelQueries({ queryKey: ['locations'] });
    const previous = queryClient.getQueryData(['locations']);
    queryClient.setQueryData(['locations'], (old) => [...old, newData]);
    return { previous };
  },
  onError: (err, newData, context) => {
    queryClient.setQueryData(['locations'], context.previous);
  }
});

Security Best Practices

  • Never trust client-side checks: Always use RLS policies for authorization
  • Use helper functions: Centralize role checks in database functions
  • Validate on every request: Each query goes through RLS policies
  • Avoid exposing user IDs: Use RLS auth.uid() instead

Distance Calculation

export function calculateDistance(
  lat1: number, lng1: number,
  lat2: number, lng2: number
): number {
  const R = 3959; // Earth's radius in miles
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  
  const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) *
    Math.sin(dLng/2) * Math.sin(dLng/2);
  
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
  return R * c;
}

Uses Haversine formula (Earth radius: 3958.8 miles):

// src/lib/distance.ts
export function calculateDistance(
  lat1: number, lng1: number,
  lat2: number, lng2: number
): number {
  const R = 3958.8; // Earth radius in miles
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  
  const a = Math.sin(dLat/2) ** 2 +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) *
    Math.sin(dLng/2) ** 2;
  
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

Overlap Detection

Automatic detection of overlapping store zones with recommendations.

Radius Overlap Calculation

import { checkStoreZoneOverlap, calculateDistance } from '@/hooks/useStoreZones';

function checkRadiusOverlap(t1: StoreZone, t2: StoreZone): boolean {
  if (t1.territory_type !== 'radius' || t2.territory_type !== 'radius') {
    return false;
  }
  
  // Calculate Haversine distance between centers
  const distance = calculateDistance(
    t1.center_lat!, t1.center_lng!,
    t2.center_lat!, t2.center_lng!
  );
  
  // Overlap if distance < sum of radii
  const combinedRadius = (t1.radius_miles || 0) + (t2.radius_miles || 0);
  return distance < combinedRadius;
}

// Haversine formula (from src/lib/distance.ts)
function calculateDistance(lat1: number, lng1: number, lat2: number, lng2: number): number {
  const R = 3958.8; // Earth's radius in miles
  const dLat = toRad(lat2 - lat1);
  const dLng = toRad(lng2 - lng1);
  
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
            Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) *
            Math.sin(dLng / 2) * Math.sin(dLng / 2);
  
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

Zip Code Overlap Detection

function checkZipCodeOverlap(t1: StoreZone, t2: StoreZone): boolean {
  if (t1.territory_type !== 'zipcode' || t2.territory_type !== 'zipcode') {
    return false;
  }
  
  const zips1 = new Set(t1.zip_codes || []);
  const zips2 = t2.zip_codes || [];
  
  // Overlap if any zip code appears in both store zones
  return zips2.some(zip => zips1.has(zip));
}

Troubleshooting

Common Issues

Maps Not Loading

Symptoms: Blank map area, "Loading..." spinner never stops

  • Check GOOGLE_MAPS_API_KEY secret is configured
  • Verify API key has Maps JavaScript API, Places API, and Geocoding API enabled
  • Check browser console for API errors
  • Ensure key has appropriate domain restrictions

Authentication Issues

Symptoms: Can't log in, session expires immediately

  • Verify auto-confirm email is enabled in auth settings
  • Check if user has entry in user_roles table
  • Clear browser storage and cookies
  • Check auth configuration in backend

RLS Policy Errors

Symptoms: "new row violates row-level security policy"

  • Ensure user_id is set to auth.uid() in insert
  • Verify user is authenticated before mutation
  • Check policy conditions match your data
  • For org data, ensure user has valid user_roles entry

Missing Data

Symptoms: Data exists in DB but doesn't appear in UI

  • Check is_active filter on queries
  • Verify RLS policies allow read access
  • Check query limits (default 1000 rows)
  • Invalidate query cache: queryClient.invalidateQueries()

Slow Performance

Symptoms: Long load times, laggy interactions

  • Check network tab for slow API calls
  • Verify indexes exist on frequently queried columns
  • Use React DevTools Profiler to find re-render issues
  • Check for N+1 query patterns in hooks
